31[[nodiscard]] auto
32compute_total_kinetic_energy(const RigidBodySOA& bodies, bool include_angular) noexcept
33 -> f32
34{
35 auto E = 0.0f;
36 for (auto i = 0zu; i < bodies.size(); ++i)
37 {
38 if (bodies.inv_masses[i] <= 0.0f)
39 {
40 continue;
41 }
42 E += 0.5f * glm::dot(bodies.velocities[i], bodies.velocities[i]) / bodies.inv_masses[i];
43 if (include_angular)
44 {
45 E += 0.5f * glm::dot(
46 bodies.angular_velocities[i], bodies.inertia_worlds[i] * bodies.angular_velocities[i]
47 );
48 }
49 }
50 return E;
51}

155auto PhysicsContext::step() -> void
156{
157 contact_arena.clear();
158 // How long the frame lasted, we do physics processing
159 // until physics time reaches that time.
160 const Duration dt{time_step};
161 const f32 dt_s{dt_f32(dt)};
162
163 update_inv_inertia_world(bodies_soa);
164
165 parallel_for_index(
166 bodies_soa.size(),
167 [&](usize i) -> void
168 {
169 for (const auto& force : simple_forces)
170 {
171 apply_force(bodies_soa, i, force);
172 }
173 }
174 );
175 for (auto& force : complex_forces)
176 {
177 apply_force(bodies_soa, force);
178 }
179
180 integrate_forces(bodies_soa, dt_s);
181 integrate_velocities(bodies_soa, dt_s);
182 debug_collision_stats =
183 generate_obb_contacts(bodies_soa, contact_arena, collision_scratch, debug_tracking_enabled);
184
185 if (debug_tracking_enabled)
186 { // Setup debug info for this step
187 debug_contacts.clear();
188 debug_contacts.reserve(contact_arena.used() / sizeof(Contact));
189 for (const auto& contact : contact_arena.as_span<Contact>())
190 {
191 debug_contacts.push_back(
192 DebugContact{
193 .a_id = bodies_soa.ids[contact.a_idx],
194 .b_id = bodies_soa.ids[contact.b_idx],
195 .p = contact.p,
196 .n = contact.n,
197 .penetration = contact.penetration,
198 .allow_warm_start = contact.allow_warm_start,
199 }
200 );
201 }
202 }
203 else
204 {
205 debug_contacts.clear();
206 debug_collision_stats = {};
207 }
208
209 // Pull warm-start state from cache into contacts before warm start + solve.
210 for (auto& contact : contact_arena.as_span<Contact>())
211 {
212 if (!contact.allow_warm_start)
213 {
214 continue;
215 }
216 const ContactKey key =
217 make_contact_key(bodies_soa.ids[contact.a_idx], bodies_soa.ids[contact.b_idx], contact.p);
218 if (auto it = contact_cache.find(key); it != contact_cache.end())
219 {
220 contact.lambda_n = it->second.lambda_n;
221 contact.lambda_t = it->second.lambda_t;
222 contact.t_hat = it->second.t_hat;
223 contact.has_t_hat = it->second.has_t_hat;
224 }
225 }
226
227 for (auto& contact : contact_arena.as_span<Contact>())
228 {
229 if (!contact.allow_warm_start)
230 {
231 continue;
232 }
233 if (contact.penetration < 0.05f)
234 {
235 warm_start_contact(bodies_soa, contact);
236 }
237 }
238
239 solve_velocity_constraints(bodies_soa, contact_arena, dt_s);
240 solve_position_constraints(bodies_soa, contact_arena);
241
242 contact_cache.clear();
243 contact_cache.reserve((contact_arena.used() / sizeof(Contact)) * 2zu);
244
245 for (const auto& contact : contact_arena.as_span<Contact>())
246 {
247 if (!contact.allow_warm_start)
248 {
249 continue;
250 }
251 const ContactKey key =
252 make_contact_key(bodies_soa.ids[contact.a_idx], bodies_soa.ids[contact.b_idx], contact.p);
253
254 contact_cache.insert_or_assign(
255 key,
256 ContactCacheEntry{
257 .lambda_n = contact.lambda_n,
258 .lambda_t = contact.lambda_t,
259 .t_hat = contact.t_hat,
260 .has_t_hat = contact.has_t_hat,
261 }
262 );
263 }
264
265 apply_sleep_and_damping(bodies_soa, dt_s);
266 clear_accumulators(bodies_soa);
267
268 if (debug_tracking_enabled)
269 {
270 debug_total_kinetic_energy = compute_total_kinetic_energy(bodies_soa, true);
271
272 debug_energy_sample_accum += dt;
273 while (debug_energy_sample_accum >= Duration{k_energy_sample_dt})
274 {
275 debug_energy_sample_accum -= Duration{k_energy_sample_dt};
276 debug_total_kinetic_energy_history.push(debug_total_kinetic_energy);
277 }
278 }
279 else
280 {
281 debug_total_kinetic_energy = 0.0f;
282 debug_energy_sample_accum = Duration{0.0};
283 debug_total_kinetic_energy_history.clear();
284 }
285 time = time + std::chrono::duration_cast<Clock::duration>(dt);
286}
