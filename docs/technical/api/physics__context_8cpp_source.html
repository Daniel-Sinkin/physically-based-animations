31[[nodiscard]] auto
32compute_total_kinetic_energy(const RigidBodySOA& bodies, bool include_angular) noexcept
33 -> f32
34{
35 auto E = 0.0f;
36 for (auto i = 0zu; i < bodies.size(); ++i)
37 {
38 if (bodies.inv_masses[i] <= 0.0f)
39 {
40 continue;
41 }
42 E += 0.5f * glm::dot(bodies.velocities[i], bodies.velocities[i]) / bodies.inv_masses[i];
43 if (include_angular)
44 {
45 // TODO: Cache this directly instead of re-inverting every time
46 const glm::mat3 I_world = glm::inverse(bodies.inv_inertia_worlds[i]);
47 E += 0.5f * glm::dot(
48 bodies.angular_velocities[i], I_world * bodies.angular_velocities[i]
49 );
50 }
51 }
52 return E;
53}

153auto PhysicsContext::step() -> void
154{
155 contact_arena.clear();
156 // How long the frame lasted, we do physics processing
157 // until physics time reaches that time.
158 const Duration dt{time_step};
159 const f32 dt_s{dt_f32(dt)};
160
161 update_inv_inertia_world(bodies_soa);
162
163 parallel_for_index(
164 bodies_soa.size(),
165 [&](usize i) -> void
166 {
167 for (const auto& force : simple_forces)
168 {
169 apply_force(bodies_soa, i, force);
170 }
171 }
172 );
173 for (auto& force : complex_forces)
174 {
175 apply_force(bodies_soa, force);
176 }
177
178 integrate_forces(bodies_soa, dt_s);
179 integrate_velocities(bodies_soa, dt_s);
180 debug_collision_stats =
181 generate_obb_contacts(bodies_soa, contact_arena, collision_scratch, debug_tracking_enabled);
182
183 if (debug_tracking_enabled)
184 { // Setup debug info for this step
185 debug_contacts.clear();
186 debug_contacts.reserve(contact_arena.used() / sizeof(Contact));
187 for (const auto& contact : contact_arena.as_span<Contact>())
188 {
189 debug_contacts.push_back(
190 DebugContact{
191 .a_id = bodies_soa.ids[contact.a_idx],
192 .b_id = bodies_soa.ids[contact.b_idx],
193 .p = contact.p,
194 .n = contact.n,
195 .penetration = contact.penetration,
196 .allow_warm_start = contact.allow_warm_start,
197 }
198 );
199 }
200 }
201 else
202 {
203 debug_contacts.clear();
204 debug_collision_stats = {};
205 }
206
207 // Pull warm-start state from cache into contacts before warm start + solve.
208 for (auto& contact : contact_arena.as_span<Contact>())
209 {
210 if (!contact.allow_warm_start)
211 {
212 continue;
213 }
214 const ContactKey key =
215 make_contact_key(bodies_soa.ids[contact.a_idx], bodies_soa.ids[contact.b_idx], contact.p);
216 if (auto it = contact_cache.find(key); it != contact_cache.end())
217 {
218 contact.lambda_n = it->second.lambda_n;
219 contact.lambda_t = it->second.lambda_t;
220 contact.t_hat = it->second.t_hat;
221 contact.has_t_hat = it->second.has_t_hat;
222 }
223 }
224
225 for (auto& contact : contact_arena.as_span<Contact>())
226 {
227 if (!contact.allow_warm_start)
228 {
229 continue;
230 }
231 if (contact.penetration < 0.05f)
232 {
233 warm_start_contact(bodies_soa, contact);
234 }
235 }
236
237 solve_velocity_constraints(bodies_soa, contact_arena, dt_s);
238 solve_position_constraints(bodies_soa, contact_arena);
239
240 contact_cache.clear();
241 contact_cache.reserve((contact_arena.used() / sizeof(Contact)) * 2zu);
242
243 for (const auto& contact : contact_arena.as_span<Contact>())
244 {
245 if (!contact.allow_warm_start)
246 {
247 continue;
248 }
249 const ContactKey key =
250 make_contact_key(bodies_soa.ids[contact.a_idx], bodies_soa.ids[contact.b_idx], contact.p);
251
252 contact_cache.insert_or_assign(
253 key,
254 ContactCacheEntry{
255 .lambda_n = contact.lambda_n,
256 .lambda_t = contact.lambda_t,
257 .t_hat = contact.t_hat,
258 .has_t_hat = contact.has_t_hat,
259 }
260 );
261 }
262
263 apply_sleep_and_damping(bodies_soa, dt_s);
264 clear_accumulators(bodies_soa);
265
266 if (debug_tracking_enabled)
267 {
268 debug_total_kinetic_energy = compute_total_kinetic_energy(bodies_soa, true);
269
270 debug_energy_sample_accum += dt;
271 while (debug_energy_sample_accum >= Duration{k_energy_sample_dt})
272 {
273 debug_energy_sample_accum -= Duration{k_energy_sample_dt};
274 debug_total_kinetic_energy_history.push(debug_total_kinetic_energy);
275 }
276 }
277 else
278 {
279 debug_total_kinetic_energy = 0.0f;
280 debug_energy_sample_accum = Duration{0.0};
281 debug_total_kinetic_energy_history.clear();
282 }
283 time = time + std::chrono::duration_cast<Clock::duration>(dt);
284}
