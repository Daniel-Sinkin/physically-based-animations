119[[nodiscard]] inline auto contains(const AABB& aabb, const Pos3& p) noexcept -> bool
120{
121 auto x_axis_overlap = p.x >= aabb.min.x && p.x <= aabb.max.x;
122 auto y_axis_overlap = p.y >= aabb.min.y && p.y <= aabb.max.y;
123 auto z_axis_overlap = p.z >= aabb.min.z && p.z <= aabb.max.z;
