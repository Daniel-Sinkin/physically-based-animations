39auto EngineContext::maybe_emit_cube(const TimePoint& now) -> void
40{
41 if (gfx.editor.grab.active || gfx.imgui_uses_keyboard || !gfx.viewport_image_hovered
42 || !editor_input.key_down(EditorKey::F))
43 {
44 spit_cube.has_last_emit = false;
45 return;
46 }
47
48 const auto spawn_interval = Duration{k_emitter_spawn_interval_s};
49 if (spit_cube.has_last_emit && (now - spit_cube.last_emit) < spawn_interval)
50 {
51 return;
52 }
53
54 spit_cube.last_emit = now;
55 spit_cube.has_last_emit = true;
56
57 auto& cam = simulation.world.editor_state().camera();
58
59 const auto cam_pos = cam.position();
60 auto forward = cam.pivot - cam_pos;
61 const auto len2 = glm::dot(forward, forward);
62 if (len2 > 1e-10f)
63 {
64 forward /= std::sqrt(len2);
65 }
66 else
67 {
68 forward = k_axis_x;
69 }
70
71 const auto spawn_pos = cam_pos + k_emitter_spawn_offset * forward;
72 const auto vel = k_emitter_launch_speed * forward;
73
74 auto& e = simulation.spawn_cube(
75 spawn_pos,
76 Dir3{0.5f, 0.5f, 0.5f},
77 1.0f,
78 vel,
79 k_quaternion_identity,
80 k_zero_dir,
81 Color3{0.93f, 0.93f, 0.98f},
82 "Spit Cube"
83 );
84 if (!e.body)
85 {
86 return;
87 }
88
89 auto rb = simulation.physics.try_body(*e.body);
90 if (!rb)
91 {
92 return;
93 }
94
95 rb->grabbed = true;
96 rb->asleep = false;
97 rb->velocity = vel;
98
99 const auto disable_for =
100 std::chrono::duration_cast<Clock::duration>(Duration{k_emitter_disable_physics_s});
101 spit_cube.pending.push_back(SpitCubePending{
102 .body = *e.body,
103 .reenable_time = now + disable_for,
104 });

107auto EngineContext::update_pending_spit_cubes(const TimePoint& now, Duration frame_dt) -> void
108{
109 auto write = 0zu;
110 const auto dt_s = static_cast<f32>(frame_dt.count());
111
112 for (auto i = 0zu; i < spit_cube.pending.size(); ++i)
113 {
114 const auto pending = spit_cube.pending[i];
115 auto rb = simulation.physics.try_body(pending.body);
116 if (!rb)
117 {
118 continue;
119 }
120
121 if (rb->grabbed)
122 {
123 rb->position += rb->velocity * dt_s;
124 (void) simulation.world.set_position(rb->id, rb->position);
125 }
126
127 if (now >= pending.reenable_time)
128 {
129 rb->grabbed = false;
130 continue;
131 }
132
133 spit_cube.pending[write++] = pending;
134 }
135 spit_cube.pending.resize(write);
136}

156auto EngineContext::run() -> void
157{
158 const Duration fixed_dt{simulation.physics.time_step};
159 const Duration max_frame_dt{0.25};
160
161 reset_simulation_clock();
162
163 bool prev_paused{paused};
164
165 while (is_active())
166 {
167 { // See shutdown.hpp for details on our signal handling
168 if (g_request_close_sig)
169 {
170 g_request_close.store(true, std::memory_order_relaxed);
171 g_request_close_sig = 0;
172 }
173 if (g_request_close.load(std::memory_order_relaxed))
174 {
175 request_close();
176 }
177 }
178
179 glfwPollEvents();
180 editor_input.update(not_null{gfx.window});
181 gfx.step(editor_input);
182
183 if (!gfx.imgui_uses_keyboard && editor_input.key_pressed(EditorKey::Space))
184 {
185 set_paused(!paused);
186 }
187
188 if (paused != prev_paused)
189 {
190 if (paused)
191 {
192 ui_log("Paused (SPACE to resume)");
193 }
194 else
195 {
196 ui_log("Running (SPACE to pause)");
197 }
198 prev_paused = paused;
199 }
200
201 const TimePoint now{Clock::now()};
202 Duration frame_dt{std::chrono::duration_cast<Duration>(now - frame_time)};
203 frame_time = now;
204 frame_dt = std::min(frame_dt, max_frame_dt);
205
206 update_pending_spit_cubes(now, frame_dt);
207 maybe_emit_cube(now);
208
209 if (paused)
210 {
211 continue;
212 }
213
214 accumulator += frame_dt;
215 while (accumulator >= fixed_dt)
216 {
217 simulation.physics.step();
218 accumulator -= fixed_dt;
219 }
220
221 simulation.sync_physics_to_world();
222 }
223}
