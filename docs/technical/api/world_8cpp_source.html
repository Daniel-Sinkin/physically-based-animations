158auto World::remove_entity(EntityId id) noexcept -> void
159{
160 auto it = id_to_index_.find(id);
161 if (it == id_to_index_.end())
162 {
163 return;
164 }
165
166 const auto idx = it->second;
167 const auto last = entities_.empty() ? 0zu : (entities_.size() - 1zu);
168
169 editor_state_.erase_from_selection(id);
170 id_to_index_.erase(it);
171
172 const auto transform_ok = transforms_.swap_erase(idx);
173 if (!transform_ok)
174 {
175 std::println(stderr, "TransformSOA erase failed for entity {}", id);
176 std::terminate();
177 }
178
179 if (idx != last)
180 {
181 entities_[idx] = std::move(entities_[last]);
182 id_to_index_.insert_or_assign(entities_[idx].id, idx);
183 }
184
185 entities_.pop_back();
186}
