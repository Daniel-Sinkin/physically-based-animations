276struct Contact
277{
278 usize a_idx{k_invalid_idx};
279 usize b_idx{k_invalid_idx};
280
281 Pos3 p{}; // contact point (world)
282 Dir3 n{k_axis_z}; // unit normal (world), direction is b -> a
283 f32 penetration{}; // >= 0
284 f32 lambda_n{}; // accumulated normal impulse
285 f32 lambda_t{}; // accumulated friction impulse
286
287 Dir3 t_hat{}; // tangent direction
288 bool has_t_hat{false}; // do we have a cached tangent direction?
289 bool allow_warm_start{true};
290
291 [[nodiscard]] auto validate() const -> ContactValidity
292 {
293 const auto finite_f32 = [](f32 x) noexcept -> bool
294 { return std::isfinite(static_cast<f64>(x)); };
295
296 const auto finite_v3 = [&](const glm::vec3& v) noexcept -> bool
297 { return finite_f32(v.x) && finite_f32(v.y) && finite_f32(v.z); };
298
299 if (a_idx == k_invalid_idx)
300 {
301 std::println(
302 stderr,
303 "Contact invalid: {} (a_idx is k_invalid_idx)",
304 to_string(ContactValidity::InvalidAId)
305 );
306 return ContactValidity::InvalidAId;
307 }
308 if (b_idx == k_invalid_idx)
309 {
310 std::println(
311 stderr,
312 "Contact invalid: {} (b_idx is k_invalid_idx)",
313 to_string(ContactValidity::InvalidBId)
314 );
315 return ContactValidity::InvalidBId;
316 }
317
318 if (!finite_v3(p))
319 {
320 std::println(
321 stderr,
322 "Contact invalid: {} (p=({}, {}, {}))",
323 to_string(ContactValidity::NonFinitePoint),
324 static_cast<f64>(p.x),
325 static_cast<f64>(p.y),
326 static_cast<f64>(p.z)
327 );
328 return ContactValidity::NonFinitePoint;
329 }
330
331 if (!finite_v3(n))
332 {
333 std::println(
334 stderr,
335 "Contact invalid: {} (n=({}, {}, {}))",
336 to_string(ContactValidity::NonFiniteNormal),
337 static_cast<f64>(n.x),
338 static_cast<f64>(n.y),
339 static_cast<f64>(n.z)
340 );
341 return ContactValidity::NonFiniteNormal;
342 }
343
344 if (!finite_f32(penetration))
345 {
346 std::println(
347 stderr,
348 "Contact invalid: {} (penetration={})",
349 to_string(ContactValidity::NonFinitePenetration),
350 static_cast<f64>(penetration)
351 );
352 return ContactValidity::NonFinitePenetration;
353 }
354
355 if (penetration < 0.0f)
356 {
357 std::println(
358 stderr,
359 "Contact invalid: {} (penetration={})",
360 to_string(ContactValidity::NegativePenetration),
361 static_cast<f64>(penetration)
362 );
363 return ContactValidity::NegativePenetration;
364 }
365
366 const auto len2 = glm::dot(n, n);
367 if (!finite_f32(len2) || len2 <= 1e-12f)
368 {
369 std::println(
370 stderr,
371 "Contact invalid: {} (|n|^2={})",
372 to_string(ContactValidity::NonUnitNormal),
373 static_cast<f64>(len2)
374 );
375 return ContactValidity::NonUnitNormal;
376 }
377
378 const auto len = std::sqrt(len2);
379 const auto err = std::abs(len - 1.0f);
380 if (err > 1e-3f)
381 {
382 std::println(
383 stderr,
384 "Contact invalid: {} (|n|={}, err={})",
385 to_string(ContactValidity::NonUnitNormal),
386 static_cast<f64>(len),
387 static_cast<f64>(err)
388 );
389 return ContactValidity::NonUnitNormal;
390 }
391
392 return ContactValidity::Ok;
393 }
