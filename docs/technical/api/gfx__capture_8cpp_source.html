21auto GfxContext::capture_viewport_rgba8(std::vector<u8>& out) const -> bool
22{
23 Expects(loaded_glad);
24 if (!loaded_glad)
25 {
26 std::println("Glad not loaded");
27 return false;
28 }
29 if (!viewport_fb_rect_valid)
30 {
31 std::println("Viewport framebuffer not valid");
32 return false;
33 }
34 if (viewport_fbo.fbo == 0 || viewport_fbo.color_tex == 0)
35 {
36 std::println("Frame buffer fbo or color_tex is invalid");
37 return false;
38 }
39
40 const int w{viewport_fbo.width};
41 const int h{viewport_fbo.height};
42 if (w <= 0 || h <= 0)
43 {
44 std::println("Viewport is empty");
45 return false;
46 }
47 const usize num_bytes = narrow_cast<usize>(w) * narrow_cast<usize>(h) * 4zu;
48 out.resize(num_bytes);
49
50 GLint prev_fbo{0};
51 glGetIntegerv(GL_FRAMEBUFFER_BINDING, &prev_fbo);
52 GLint prev_read_buffer{0};
53 glGetIntegerv(GL_READ_BUFFER, &prev_read_buffer);
54 GLint prev_pack_alignment{0};
55 glGetIntegerv(GL_PACK_ALIGNMENT, &prev_pack_alignment);
56
57 glBindFramebuffer(GL_FRAMEBUFFER, viewport_fbo.fbo);
58 glReadBuffer(GL_COLOR_ATTACHMENT0);
59 glPixelStorei(GL_PACK_ALIGNMENT, 1);
60
61 glReadPixels(0, 0, w, h, GL_RGBA, GL_UNSIGNED_BYTE, out.data());
62
63 glBindFramebuffer(GL_FRAMEBUFFER, static_cast<GLuint>(prev_fbo));
64 glReadBuffer(static_cast<GLenum>(prev_read_buffer));
65 glPixelStorei(GL_PACK_ALIGNMENT, prev_pack_alignment);
66
67 const GLenum err = glGetError();
68 return err == GL_NO_ERROR;
69}
