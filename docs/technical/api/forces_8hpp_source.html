67inline auto apply_force(RigidBodySOA& bodies, usize body_idx, const SimpleForce& force) noexcept
68 -> void
69{
70 if (
71 bodies.is_static(body_idx) || bodies.is_asleep(body_idx) || bodies.is_grabbed(body_idx)
72 )
73 {
74 return;
75 }
76
77 const auto inv_mass = bodies.inv_masses[body_idx];
78 auto& force_accum = bodies.force_accums[body_idx];
79 auto& torque_accum = bodies.torque_accums[body_idx];
80
81 std::visit(
82 overloaded{
83 [&](const GravityForce& g) noexcept { force_accum += g.accel / inv_mass; },
84 [&](const AttractorForce& a) noexcept
85 {
86 const auto min_r = std::max(a.min_radius, 1e-6f);
87 const auto min_r2 = min_r * min_r;
88
89 const auto d = a.target - bodies.positions[body_idx];
90 const auto d2 = glm::dot(d, d);
91 if (d2 <= min_r2)
92 {
93 return;
94 }
95
96 const auto inv_len = 1.0f / std::sqrt(d2);
97 const auto dir = d * inv_len;
98
99 const auto accel = a.magnitude * dir;
100 force_accum += accel / inv_mass;
101 },
102 [&](const RepulsionForce& r) noexcept
103 {
104 const auto min_r = std::max(r.min_radius, 1e-6f);
105 const auto min_r2 = min_r * min_r;
106 const auto range = std::max(r.range, min_r);
107
108 const auto d = bodies.positions[body_idx] - r.target;
109 const auto d2 = glm::dot(d, d);
110 if (d2 <= min_r2)
111 {
112 return;
113 }
114
115 const auto dist = std::sqrt(d2);
116 if (dist >= range)
117 {
118 return;
119 }
120
121 const auto dir = d / dist;
122
123 const auto t = (range - dist) / (range - min_r);
124 const auto accel_mag = std::clamp(t, 0.0f, 1.0f) * r.accel_max;
125
126 const auto accel = accel_mag * dir;
127 force_accum += accel / inv_mass;
128 },
129 [&](const MotorForce& m) noexcept
130 {
131 if (m.id != k_invalid_id && bodies.ids[body_idx] == m.id)
132 {
133 torque_accum += m.torque;
134 }
135 },
136 },
137 force
138 );
139}

141inline auto apply_force(RigidBodySOA& bodies, const ComplexForce& force) noexcept -> void
142{
143 std::visit(
144 overloaded{
145 [&](const NBodyForce& p) noexcept
146 {
147 const auto n = bodies.size();
148 if (n < 2zu)
149 {
150 return;
151 }
152
153 const auto eps2 = p.softening * p.softening;
154
155 for (usize i{0zu}; i < n; ++i)
156 {
157 if (
158 bodies.is_static(i) || bodies.is_asleep(i) || bodies.is_grabbed(i)
159 || bodies.inv_masses[i] <= 0.0f
160 )
161 {
162 continue;
163 }
164
165 const auto m_a = 1.0f / bodies.inv_masses[i];
166
167 for (usize j{i + 1zu}; j < n; ++j)
168 {
169 if (
170 bodies.is_static(j) || bodies.is_asleep(j) || bodies.is_grabbed(j)
171 || bodies.inv_masses[j] <= 0.0f
172 )
173 {
174 continue;
175 }
176
177 const auto m_b = 1.0f / bodies.inv_masses[j];
178
179 const auto r = bodies.positions[j] - bodies.positions[i];
180 const auto r2 = glm::dot(r, r) + eps2;
181
182 const auto inv_r = 1.0f / std::sqrt(r2);
183 const auto inv_r3 = inv_r * inv_r * inv_r;
184
185 const auto F = (p.G * m_a * m_b) * r * inv_r3;
186
187 bodies.force_accums[i] += F;
188 bodies.force_accums[j] -= F;
189 }
190 }
191 },
192 },
193 force
194 );
195}
