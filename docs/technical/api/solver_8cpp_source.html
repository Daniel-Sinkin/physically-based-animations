177auto apply_impulse_contact_friction(
178 RigidBodySOA& bodies, usize a_idx, usize b_idx, Contact& contact, Dir3 r_a, Dir3 r_b, Dir3 n
179) noexcept -> void
180{
181 auto a = body_ref(bodies, a_idx);
182 auto b = body_ref(bodies, b_idx);
183
184 {
185 Expects(!a.is_static() || !b.is_static());
186 Expects(!a.asleep || !b.asleep);
187 }
188 // This section adapted from [Catto05] (4.3) Friction Constraints.
189 // The paper explicitly uses a orthonormal tangent basis {u1, u2} with
190 // u1 x u2 = n and computes (v_rel.u1) and (v_rel.u2) seperately.
191 // To avoid computing these U:={u1, u2} explicitly this code does
192 //
193 // proj_U(v_rel) = v_rel - (v_rel.n)*n == (v_rel.u1)*u1 + (v_rel.u2)*u2
194 //
195 // which is an orthonormal projection onto the orthogonal space of the
196 // normal direction (i.e. exactly the tangent space).
197 const auto p_a_dot = a.velocity + glm::cross(a.angular_velocity, r_a);
198 const auto p_b_dot = b.velocity + glm::cross(b.angular_velocity, r_b);
199 const auto v_rel_w = p_a_dot - p_b_dot;
200
201 // Projection onto the tangent
202 // v_rel.n
203 const auto vrel_dot_n = glm::dot(v_rel_w, n);
204 // v_rel - (v_rel.n) * n
205 const auto v_t = v_rel_w - vrel_dot_n * n;
206 // Equation (21) would be C_{u1}' = v_rel_w . u1
207 // Equation (22) would be C_{u2}' = v_rel_w . u2
208
209 // Model friction as a force acting in the opposite direction of "slip"
210 const auto vt2 = glm::dot(v_t, v_t);
211 auto t_hat = Dir3{};
212 if (vt2 > 1e-12f)
213 {
214 t_hat = v_t / std::sqrt(vt2);
215
216 contact.t_hat = t_hat;
217 contact.has_t_hat = true;
218 }
219 else
220 {
221 if (!contact.has_t_hat)
222 {
223 return;
224 }
225 t_hat = contact.t_hat;
226 }
227
228 // These are the angular Jacobian terms (r x u) from Equation (23)
229 const auto r_a_cross_t = glm::cross(r_a, t_hat);
230 const auto r_b_cross_t = glm::cross(r_b, t_hat);
231
232 // Change in angular velocity is given by
233 //
234 // Delta omega = I^-1(r x J)
235 //
236 const auto invI_r_axt = a.inv_inertia_world * r_a_cross_t;
237 const auto invI_r_bxt = b.inv_inertia_world * r_b_cross_t;
238
239 // Computes effective mass along t_hat, often denoted by just k
240 const auto k2a = a.inv_mass + glm::dot(t_hat, glm::cross(invI_r_axt, r_a));
241 const auto k2b = b.inv_mass + glm::dot(t_hat, glm::cross(invI_r_bxt, r_b));
242 const auto effective_mass = k2a + k2b;
243 if (effective_mass <= 1e-12f)
244 {
245 return;
246 }
247
248 const auto old_lambda_t = contact.lambda_t;
249
250 auto delta_lambda_t = -glm::dot(v_rel_w, t_hat) / effective_mass;
251 const auto max_jt = k_friction * contact.lambda_n;
252 if (max_jt <= 0.0f)
253 {
254 contact.lambda_t = 0.0f;
255 return;
256 }
257
258 const auto new_lambda_t = std::clamp(old_lambda_t + delta_lambda_t, -max_jt, +max_jt);
259 delta_lambda_t = new_lambda_t - old_lambda_t;
260 contact.lambda_t = new_lambda_t;
261
262 const auto friction_impulse = delta_lambda_t * t_hat;
263
264 if (!a.is_static())
265 {
266 a.velocity += friction_impulse * a.inv_mass;
267 a.angular_velocity += a.inv_inertia_world * glm::cross(r_a, friction_impulse);
268 }
269 if (!b.is_static())
270 {
271 b.velocity -= friction_impulse * b.inv_mass;
272 b.angular_velocity -= b.inv_inertia_world * glm::cross(r_b, friction_impulse);
273 }
274}

276auto apply_impulse_contact(
277 RigidBodySOA& bodies, usize a_idx, usize b_idx, Contact& contact, f32 restitution, f32 dt_s
278) noexcept -> void
279{
280 auto a = body_ref(bodies, a_idx);
281 auto b = body_ref(bodies, b_idx);
282
283 {
284 Expects(!a.is_static() || !b.is_static());
285 Expects(!a.asleep || !b.asleep);
286 Expects(restitution >= 0.0f && restitution <= 1.0f);
287 Expects(std::abs(glm::length(contact.n) - 1.0f) < 1e-5f);
288 }
289 // This is based on [Baraff97]
290
291 // "The quantity eps is called the coefficient of restitution (...)"
292 constexpr f32 k_restitution_threshold{2.0f};
293 auto eps = restitution;
294 // (...) and must satisfy 0 <= eps <= 1. If eps = 1 then v_rel^+ = -v_rel^-,
295 // and the collision is perfectly 'bouncy'; in particular, no kinetic energy is lost.
296 // At the other end of the spectrum, eps = 0 results in v_rel^+ = 0, and a maxium
297 // of kinetic energy is lost.
298
299 // n^(t_0) the unit surface normal. Points from b to a
300 const auto n = contact.n;
301
302 // Contact Point
303 const auto p = contact.p;
304
305 // Center of mass
306 const auto x_a = a.position;
307 const auto x_b = b.position;
308
309 // (Linear) Velocity
310 const auto v_a = a.velocity;
311 const auto v_b = b.velocity;
312
313 // Displacement from center of mass
314 const auto r_a = p - x_a;
315 const auto r_b = p - x_b;
316
317 // Angular velocities are denoted by omega_a and omega_b respectively
318 // (8-1) p_a'(t_0) = v_a(t_0) + omega_a(t_0) x (p_a(t_0) - x_a(t_0))
319 const auto p_a_dot = v_a + glm::cross(a.angular_velocity, r_a);
320 // (8-2) p_b'(t_0) = v_b(t_0) + omega_b(t_0) x (p_b(t_0) - x_b(t_0))
321 const auto p_b_dot = v_b + glm::cross(b.angular_velocity, r_b);
322
323 // (8-3) v_rel = n^ . (p_a'(t_0) - p_b'(t_0)) // Normal Relative Velocity
324 const auto v_rel = glm::dot(n, p_a_dot - p_b_dot);
325
326 // Impulse is given by J = j * n^ where we can derive j based on constraints.
327 // Will omit the derivation, can express the v_a^+ and v_a^- as well as omega_a^+ and
328 // omega_a^- conditions both via coefficient of restitution as well as the impulse identity
329 // and solve for the magnitude j
330
331 // Angular impulse direction
332 const auto r_a_cross_n = glm::cross(r_a, n);
333 const auto r_b_cross_n = glm::cross(r_b, n);
334 const auto invI_r_axn = a.inv_inertia_world * r_a_cross_n;
335 const auto invI_r_bxn = b.inv_inertia_world * r_b_cross_n;
336
337 // (8-18) Denominator
338 // "Effective Mass" denominator; in the literature often denoted by k
339 const auto k_a = a.inv_mass + glm::dot(n, glm::cross(invI_r_axn, r_a));
340 const auto k_b = b.inv_mass + glm::dot(n, glm::cross(invI_r_bxn, r_b));
341 const auto effective_mass = k_a + k_b;
342 if (effective_mass <= 1e-12f)
343 {
344 return;
345 }
346 if (-v_rel < k_restitution_threshold)
347 {
348 eps = 0.0f;
349 }
350
351 // delta_lambda_n is accumulation of what is j (impulse magnitude) in the paper
352 const auto old_lambda_n = contact.lambda_n;
353
354 constexpr f32 k_max_bias_speed{2.0f};
355 const auto effective_pen = std::max(0.0f, contact.penetration - k_pen_tolerance);
356 const auto bias_raw = -(k_pen_correction_frag / dt_s) * effective_pen;
357 const auto bias = std::clamp(bias_raw, -k_max_bias_speed, 0.0f);
358 auto delta_lambda_n = -(((1.0f + eps) * v_rel) + bias) / effective_mass;
359
360 const auto new_lambda_n = std::max(old_lambda_n + delta_lambda_n, 0.0f);
361 delta_lambda_n = new_lambda_n - old_lambda_n;
362 contact.lambda_n = new_lambda_n;
363 if (delta_lambda_n != 0.0f)
364 {
365 const auto impulse = delta_lambda_n * n;
366
367 // Equation (8-5) Delta v = J / M
368 // Equation (8-6) tau_impulse = (p - x(t)) x J
369 // "The change in angular velocity is simply I^{-1}(t_0)\tau_{impulse}"
370 if (!a.is_static())
371 {
372 a.velocity += impulse * a.inv_mass;
373 const auto tau_a_impulse = glm::cross(r_a, impulse);
374 a.angular_velocity += a.inv_inertia_world * tau_a_impulse;
375 }
376 if (!b.is_static())
377 {
378 // Recall n^(t_0) points from b to a so we have to reverse impulse direction
379 b.velocity -= impulse * b.inv_mass;
380 const auto tau_b_impulse = glm::cross(r_b, impulse);
381 b.angular_velocity -= b.inv_inertia_world * tau_b_impulse;
382 }
383 }
384 apply_impulse_contact_friction(bodies, a_idx, b_idx, contact, r_a, r_b, n);
385}

409{
410 parallel_for_index(
411 bodies.size(),
412 [&](usize i) -> void
413 {
414 auto b = body_ref(bodies, i);
415 if (b.is_static() || b.asleep || b.grabbed)
416 {
417 return;
418 }

420 const auto a = b.force_accum * b.inv_mass;

426 }
427 );

451auto warm_start_contact(RigidBodySOA& bodies, Contact& contact) noexcept -> void
452{
453
454 auto a = body_ref(bodies, contact.a_idx);
455 auto b = body_ref(bodies, contact.b_idx);
456
457 if (a.is_static() && b.is_static())
458 {
459 return;
460 }
461 if (a.asleep || b.asleep || a.grabbed || b.grabbed)
462 {
463 return;
464 }
465
466 const auto normal = contact.n;
467 const auto r_a = contact.p - a.position;
468 const auto r_b = contact.p - b.position;
469
470 constexpr auto warmstart_scale = 1.0f;
471 contact.lambda_n = std::clamp(contact.lambda_n, 0.0f, 50.0f);
472 contact.lambda_t = std::clamp(contact.lambda_t, -50.0f, 50.0f);
473 if (contact.lambda_n > 0.0f)
474 {
475 const auto Jn = warmstart_scale * contact.lambda_n * normal;
476
477 if (!a.is_static())
478 {
479 a.velocity += Jn * a.inv_mass;
480 a.angular_velocity += a.inv_inertia_world * glm::cross(r_a, Jn);
481 }
482 if (!b.is_static())
483 {
484 b.velocity -= Jn * b.inv_mass;
485 b.angular_velocity -= b.inv_inertia_world * glm::cross(r_b, Jn);
486 }
487 }
488
489 if (contact.lambda_t != 0.0f && contact.has_t_hat)
490 {
491 const auto Jt = warmstart_scale * contact.lambda_t * contact.t_hat;
492
493 if (!a.is_static())
494 {
495 a.velocity += Jt * a.inv_mass;
496 a.angular_velocity += a.inv_inertia_world * glm::cross(r_a, Jt);
497 }
498 if (!b.is_static())
499 {
500 b.velocity -= Jt * b.inv_mass;
501 b.angular_velocity -= b.inv_inertia_world * glm::cross(r_b, Jt);
502 }
503 }

519auto solve_velocity_constraints(
520 RigidBodySOA& bodies, ArenaAllocator& contact_arena, f32 dt_s
521) noexcept -> void
522{
523 for (auto& contact : contact_arena.as_span<Contact>())
524 {
525 auto a = body_ref(bodies, contact.a_idx);
526 auto b = body_ref(bodies, contact.b_idx);
527
528 auto can_wake_up = [](const auto& rb) -> bool { return rb.is_static() || !rb.asleep; };
529 if (can_wake_up(a) && can_wake_up(b))
530 {
531 continue;
532 }
533
534 const auto should_wake = [&]() -> bool
535 {
536 if (contact.penetration > k_pen_tolerance)
537 {
538 return true;
539 }
540 const auto v_rel = get_relative_velocity_at_world_point(
541 body_cref(bodies, contact.a_idx), body_cref(bodies, contact.b_idx), contact.p
542 );
543 return glm::dot(contact.n, v_rel) < -0.05f;
544 }();
545 if (
546 should_wake
547 && (wake_up(body_ref(bodies, contact.a_idx)) || wake_up(body_ref(bodies, contact.b_idx)))
548 )
549 {
550 contact.lambda_n = 0.0f;
551 contact.lambda_t = 0.0f;
552 contact.has_t_hat = false;
553 }
554 }
555
556 for (auto i = 0; i < k_solver_iterations; ++i)
557 {
558 for (auto& contact : contact_arena.as_span<Contact>())
559 {
560 auto a = body_ref(bodies, contact.a_idx);
561 auto b = body_ref(bodies, contact.b_idx);
562 if (a.is_static() && b.is_static())
563 {
564 continue;
565 }
566 if (a.grabbed || b.grabbed)
567 {
568 contact.lambda_n = 0.0f;
569 contact.lambda_t = 0.0f;
570 contact.has_t_hat = false;
571 continue;
572 }
573 if (a.asleep && b.asleep)
574 {
575 continue;
576 }
577 apply_impulse_contact(
578 bodies, contact.a_idx, contact.b_idx, contact, k_restitution, dt_s
579 );
580 }
581 }
582}

584auto solve_position_constraints(RigidBodySOA& bodies, ArenaAllocator& contact_arena) noexcept -> void
585{
586 for (auto i = 0; i < k_position_iterations; ++i)
587 {
588 for (auto& contact : contact_arena.as_span<Contact>())
589 {
590 auto a = body_ref(bodies, contact.a_idx);
591 auto b = body_ref(bodies, contact.b_idx);
592
593 if ((a.is_static() && b.is_static()) || contact.penetration <= k_pen_tolerance)
594 {
595 continue;
596 }
597
598 const auto inv_mass_sum = a.inv_mass + b.inv_mass;
599 if (inv_mass_sum <= 1e-12f)
600 {
601 continue;
602 }
603
604 const auto correction = [&]()
605 {
606 const auto pen_excess = contact.penetration - k_pen_tolerance;
607 const auto mag = std::clamp(k_pen_percent * pen_excess, 0.0f, k_pen_max_correction);
608 return (mag / inv_mass_sum) * contact.n;
609 }();
610 if (!a.is_static())
611 {
612 a.position += correction * a.inv_mass;
613 }
614 if (!b.is_static())
615 {
616 b.position -= correction * b.inv_mass;
617 }
618 }
619 }

622auto apply_sleep_and_damping(RigidBodySOA& bodies, f32 dt_s) noexcept -> void
623{
624 parallel_for_index(
625 bodies.size(),
626 [&](usize i) -> void
627 {
628 auto b = body_ref(bodies, i);
629 if (b.grabbed)
630 {
631 b.velocity = Dir3{};
632 b.angular_velocity = Dir3{};
633 return;
634 }
635 if (b.is_static() || b.asleep)
636 {
637 return;
638 }
639
640 const auto v2 = glm::dot(b.velocity, b.velocity);
641 const auto w2 = glm::dot(b.angular_velocity, b.angular_velocity);
642
643 auto v_lim = k_linear_sleep_speed_threshold;
644 auto w_lim = k_angular_sleep_speed_threshold;
645 const auto velo_slow = v2 < v_lim * v_lim;
646 const auto angular_velo_slow = w2 < w_lim * w_lim;
647
