133 {
134 auto out = ctx.out();
135 *out++ = '[';
136
137 for (int row{0}; row < 4; ++row)
138 {
139 if (row != 0)
140 {
141 *out++ = '\n';

143 }
144
145 for (int col{0}; col < 4; ++col)

149 ctx.advance_to(out);
150 if (col != 3)

152 *out++ = ' ';
