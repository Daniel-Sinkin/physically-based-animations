30[[nodiscard]] auto lerp_color(const Color3& a, const Color3& b, f32 t) noexcept -> Color3
31{
32 return Color3{
33 a.r() + (b.r() - a.r()) * t,
34 a.g() + (b.g() - a.g()) * t,
35 a.b() + (b.b() - a.b()) * t,
36 };

39[[nodiscard]] auto to_u8_channel(f32 linear_value) noexcept -> u8
40{
41 const auto clamped = std::clamp(linear_value, 0.0f, 1.0f);
42 const auto scaled = static_cast<int>(std::lround(clamped * 255.0f));
43 return static_cast<u8>(std::clamp(scaled, 0, 255));

46[[nodiscard]] auto make_default_environment_map_rgba8(int width, int height) -> ImageRGBA8
47{
48 {
49 Expects(width > 0);
50 Expects(height > 0);
51 }
52 const auto width_f = static_cast<f32>(width);
53 const auto height_f = static_cast<f32>(height);
54
55 ImageRGBA8 img{};
56 img.width = width;
57 img.height = height;
58 img.channels = 4;
59
60 const auto pixel_count = static_cast<usize>(width) * static_cast<usize>(height);
61 img.pixels.resize(pixel_count * 4zu);
62
63 constexpr Color3 sky_zenith{0.08f, 0.22f, 0.45f};
64 constexpr Color3 sky_horizon{0.62f, 0.74f, 0.86f};
65 constexpr Color3 ground{0.18f, 0.17f, 0.16f};
66 const auto sun_dir = glm::normalize(Dir3{0.25f, -0.35f, 0.90f});
67
68 for (int y{0}; y < height; ++y)
69 {
70 const f32 y_f = static_cast<f32>(y);
71 const auto v = (y_f + 0.5f) / height_f;
72 const auto theta = v * k_pi;
73 const auto sin_theta = std::sin(theta);
74 const auto cos_theta = std::cos(theta);
75
76 for (int x{0}; x < width; ++x)
77 {
78 const f32 x_f = static_cast<f32>(x);
79 const auto u = (x_f + 0.5f) / width_f;
80 const auto phi = (u - 0.5f) * k_two_pi;
81
82 const auto dir = Dir3{
83 std::cos(phi) * sin_theta,
84 std::sin(phi) * sin_theta,
85 cos_theta,
86 };
87
88 const auto sky_t = std::clamp(0.5f * (dir.z + 1.0f), 0.0f, 1.0f);
89 const auto sky_curve = std::pow(sky_t, 0.65f);
90 const auto sky = lerp_color(sky_horizon, sky_zenith, sky_curve);
91
92 const auto horizon_blend = std::clamp((dir.z + 0.08f) / 0.18f, 0.0f, 1.0f);
93 auto base = lerp_color(ground, sky, horizon_blend);
94
95 const auto sun_cos = std::max(glm::dot(dir, sun_dir), 0.0f);
96 const auto sun_glow = std::pow(sun_cos, 48.0f);
97 const auto sun_core = std::pow(sun_cos, 1024.0f);
98
99 base.r() += 0.40f * sun_glow + 1.20f * sun_core;
100 base.g() += 0.33f * sun_glow + 0.95f * sun_core;
101 base.b() += 0.22f * sun_glow + 0.55f * sun_core;
102
103 const auto idx =
104 (static_cast<usize>(y) * static_cast<usize>(width) + static_cast<usize>(x)) * 4zu;
105 img.pixels[idx + 0zu] = to_u8_channel(base.r());
106 img.pixels[idx + 1zu] = to_u8_channel(base.g());
107 img.pixels[idx + 2zu] = to_u8_channel(base.b());
108 img.pixels[idx + 3zu] = 255u;
109 }
110 }
111
112 {
113 Ensures(img.valid());
114 }
115 return img;
116}
