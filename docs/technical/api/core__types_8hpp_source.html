57[[nodiscard]] inline constexpr auto round_up_aligned(usize x, usize align) noexcept -> usize
58{
59 {
60 Expects(align > 0 && "Align must be > 0");
61 }
62 if (align == 0)
63 {
64 return x;
65 }
66 auto retval = [&]() -> usize
67 {
68 const auto rem = x % align;
69 if (rem == 0)
70 {
71 return x;
72 }
73 else
74 {
75 return x + (align - rem);
76 }
77 }();
78
79 {
80 Ensures(retval % align == 0);
81 }
82 return retval;
83}
