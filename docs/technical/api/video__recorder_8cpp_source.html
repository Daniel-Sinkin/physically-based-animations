37{

145auto VideoRecorder::write_frame(std::span<const u8> rgba) -> bool
146{
147 if (!pipe)
148 {
149 return false;
150 }
151
152 const usize expected = rgba_frame_bytes(width, height);
153 if (expected == 0zu || rgba.size() != expected)
154 {
155 return false;
156 }
157
158 const auto written = std::fwrite(rgba.data(), 1, static_cast<size_t>(expected), pipe);
159 if (written != static_cast<size_t>(expected))
160 {
161 stop();
162 return false;
163 }
164 ++frames_written;
165 return true;
166}
