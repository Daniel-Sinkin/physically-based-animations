18struct EnergyHistoryRing
19{
20 std::array<f32, k_energy_history_len> values{};
21 usize write_head{0zu};
22 usize size_{0zu};
23
24 auto clear() noexcept -> void
25 {
26 write_head = 0zu;
27 size_ = 0zu;
28 }
29
30 auto push(f32 value) noexcept -> void
31 {
32 values[write_head] = value;
33 write_head = (write_head + 1zu) % values.size();
34 size_ = std::min(size_ + 1zu, values.size());
35 }
36
37 [[nodiscard]] auto size() const noexcept -> usize
38 {
39 return size_;
40 }
41
42 [[nodiscard]] auto empty() const noexcept -> bool
43 {
44 return size_ == 0zu;
45 }
46
47 [[nodiscard]] auto at(usize i) const noexcept -> f32
48 {
49 Expects(i < size_);
50 const auto oldest = (write_head + values.size() - size_) % values.size();
51 const auto idx = (oldest + i) % values.size();
52 return values[idx];
53 }
54};
