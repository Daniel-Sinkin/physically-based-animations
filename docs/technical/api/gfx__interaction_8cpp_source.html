77clamp_screen_rect_to_viewport(const ScreenRect& rect, const glm::vec2& vp_pos, const glm::vec2& vp_size)
78 noexcept -> ScreenRect
79{
80 const auto vp_min_x = vp_pos.x;
81 const auto vp_min_y = vp_pos.y;
82 const auto vp_max_x = vp_pos.x + std::max(1.0f, vp_size.x);
83 const auto vp_max_y = vp_pos.y + std::max(1.0f, vp_size.y);
84
85 ScreenRect out{
86 .min_x = std::clamp(rect.min_x, vp_min_x, vp_max_x),
87 .min_y = std::clamp(rect.min_y, vp_min_y, vp_max_y),
88 .max_x = std::clamp(rect.max_x, vp_min_x, vp_max_x),
89 .max_y = std::clamp(rect.max_y, vp_min_y, vp_max_y),
90 };
91
92 if (out.min_x > out.max_x)
93 {
94 std::swap(out.min_x, out.max_x);
95 }
96 if (out.min_y > out.max_y)
97 {
98 std::swap(out.min_y, out.max_y);
99 }
100 return out;

113[[nodiscard]] auto project_world_to_screen(
114 const Pos3& world_pos,
115 const glm::mat4& view_proj,
116 const glm::vec2& vp_pos,
117 const glm::vec2& vp_size
118) noexcept -> std::optional<glm::vec2>
119{
120 const auto clip = view_proj * glm::vec4(world_pos, 1.0f);
121 if (clip.w <= k_clip_w_epsilon)
122 {
123 return std::nullopt;
124 }
125
126 const auto ndc = glm::vec3{clip} / clip.w;
127
128 const auto x_01 = 0.5f * (ndc.x + 1.0f);
129 const auto y_01 = 0.5f * (1.0f - ndc.y);
130 const auto x = vp_pos.x + x_01 * vp_size.x;
131 const auto y = vp_pos.y + y_01 * vp_size.y;
132 return glm::vec2{x, y};
133}

135[[nodiscard]] auto cube_intersects_selection_rect(
136 const ModelMatrix& model,
137 const glm::mat4& view_proj,
138 const glm::vec2& vp_pos,
139 const glm::vec2& vp_size,
140 const ScreenRect& selection_rect
141) noexcept -> bool
142{
143 ScreenRect projected{
144 .min_x = std::numeric_limits<f32>::max(),
145 .min_y = std::numeric_limits<f32>::max(),
146 .max_x = std::numeric_limits<f32>::lowest(),
147 .max_y = std::numeric_limits<f32>::lowest(),
148 };
149
150 bool has_projected_corner{false};
151 for (const auto& local_corner : k_cube_local_corners)
152 {
153 const auto world_corner = model.transform_position(local_corner);
154 const auto screen_corner = project_world_to_screen(world_corner, view_proj, vp_pos, vp_size);
155 if (!screen_corner)
156 {
157 continue;
158 }
159
160 has_projected_corner = true;
161 projected.min_x = std::min(projected.min_x, screen_corner->x);
162 projected.min_y = std::min(projected.min_y, screen_corner->y);
163 projected.max_x = std::max(projected.max_x, screen_corner->x);
164 projected.max_y = std::max(projected.max_y, screen_corner->y);
165 }
166
167 if (!has_projected_corner)
168 {
169 return false;
170 }
171
172 return screen_rect_intersects(projected, selection_rect);
173}

175[[nodiscard]] auto gather_box_selection_ids(
176 const World& world,
177 const PhysicsContext& physics,
178 const ViewMatrix& view,
179 const ProjMatrix& proj,
180 const glm::vec2& vp_pos,
181 const glm::vec2& vp_size,
182 const ScreenRect& selection_rect
183) -> std::vector<EntityId>
184{
185 const auto view_proj = proj.m * view.m;
186 std::vector<EntityId> ids{};
187
188 const auto entities = world.entities();
189 ids.reserve(entities.size());
190
191 for (auto i = 0zu; i < entities.size(); ++i)
192 {
193 const auto& entity = entities[i];
194 if (entity.type != EntityType::Cube)
195 {
196 continue;
197 }
198
199 if (entity.body)
200 {
201 const auto rb = physics.try_body(*entity.body);
202 if (rb && rb->is_static())
203 {
204 continue;
205 }
206 }
207
208 const auto model = world.model_matrix_at(i);
209 if (!model)
210 {
211 continue;
212 }
213
214 if (cube_intersects_selection_rect(*model, view_proj, vp_pos, vp_size, selection_rect))
215 {
216 ids.push_back(entity.id);
217 }
218 }
219
220 return ids;
221}

223[[nodiscard]] auto is_cube_visible_in_rect(
224 const World& world,
225 const EntityId id,
226 const ModelMatrix& model,
227 const glm::mat4& view_proj,
228 const ViewMatrix& view,
229 const ProjMatrix& proj,
230 const glm::vec2& vp_pos,
231 const glm::vec2& vp_size,
232 const ScreenRect& selection_rect
233) -> bool
234{
235 for (const auto& local_sample : k_cube_visibility_samples)
236 {
237 const auto world_sample = model.transform_position(local_sample);
238 const auto screen_pos = project_world_to_screen(world_sample, view_proj, vp_pos, vp_size);
239 if (!screen_pos || !screen_rect_contains(selection_rect, *screen_pos))
240 {
241 continue;
242 }
243
244 const auto sample_ray = ray_from_imgui_rect(*screen_pos, vp_pos, vp_size, view, proj);
245 const auto sample_hit = raycast(world, sample_ray);
246 if (sample_hit && sample_hit->object_id == id)
247 {
248 return true;
249 }
250 }
251 return false;
252}

307auto GfxContext::hover_interaction(EditorInput& input) -> void
308{
309 {
310 Expects(viewport_fb_rect_valid && "If viewport hovered then it must be valid");
311 }
312 const auto& imgui_io = ImGui::GetIO();
313
314 auto& world = engine_context->simulation.world;
315 auto& world_editor_state = world.editor_state();
316 auto& cam = world_editor_state.camera();
317
318 if (viewport_image_hovered)
319 {
320 const f32 wheel{imgui_io.MouseWheel};
321 if (wheel != 0.0f)
322 { // Zooming
323 cam.distance *= std::exp(-wheel * k_zoom_speed);
324 cam.distance = std::clamp(cam.distance, 0.75f, 200.0f);
325 }
326
327 if (input.mouse_middle.down || input.key_down(EditorKey::R))
328 {
329 hover_interaction_holding_middle(input, cam);
330 }
331 }
332
333 auto& box_select = editor.box_select;
334 if (input.mouse_left.pressed() && viewport_image_hovered)
335 {
336 box_select.active = true;
337 box_select.dragging = false;
338 box_select.start_mouse_x = input.ui_mouse_x;
339 box_select.start_mouse_y = input.ui_mouse_y;
340 box_select.current_mouse_x = input.ui_mouse_x;
341 box_select.current_mouse_y = input.ui_mouse_y;
342 }
343
344 if (!box_select.active)
345 {
346 return;
347 }
348
349 box_select.current_mouse_x = input.ui_mouse_x;
350 box_select.current_mouse_y = input.ui_mouse_y;
351
352 const auto drag_dx = narrow_cast<f32>(box_select.current_mouse_x - box_select.start_mouse_x);
353 const auto drag_dy = narrow_cast<f32>(box_select.current_mouse_y - box_select.start_mouse_y);
354 if (!box_select.dragging)
355 {
356 const auto drag_d2 = drag_dx * drag_dx + drag_dy * drag_dy;
357 const auto threshold_d2 = k_box_select_drag_threshold_px * k_box_select_drag_threshold_px;
358 box_select.dragging = drag_d2 >= threshold_d2;
359 }
360
361 if (!input.mouse_left.released())
362 {
363 return;
364 }
365
366 const auto aspect = viewport_fbo.aspect_ratio();
367 const auto camera_view_matrix = cam.view_matrix();
368 const auto camera_proj_matrix = cam.proj_matrix(aspect);
369
370 if (!box_select.dragging)
371 {
372 if (viewport_image_hovered)
373 { // Point selection on click release.
374 const glm::vec2 mouse_pos{input.ui_mouse_x, input.ui_mouse_y};
375 const Ray mouse_ray{ray_from_imgui_rect(
376 mouse_pos,
377 viewport_img_pos,
378 viewport_img_size,
379 camera_view_matrix,
380 camera_proj_matrix
381 )};
382
383 if (const auto rc_res = raycast(world, mouse_ray); rc_res)
384 {
385 hover_interaction_selection(input, *rc_res);
386 }
387 else if (!world_editor_state.selected_ids.empty() && !input.key_down(EditorKey::Shift))
388 {
389 world_editor_state.clear_selection();
390 ui_log("Deselected all");
391 }
392 }
393
394 box_select.active = false;
395 box_select.dragging = false;
396 return;
397 }
398
399 const ScreenRect drag_rect = clamp_screen_rect_to_viewport(
400 make_screen_rect(
401 {narrow_cast<f32>(box_select.start_mouse_x), narrow_cast<f32>(box_select.start_mouse_y)},
402 {
403 narrow_cast<f32>(box_select.current_mouse_x),
404 narrow_cast<f32>(box_select.current_mouse_y),
405 }
406 ),
407 viewport_img_pos,
408 viewport_img_size
409 );
410
411 const auto drag_w = drag_rect.max_x - drag_rect.min_x;
412 const auto drag_h = drag_rect.max_y - drag_rect.min_y;
413 if (drag_w < k_box_select_min_axis_px && drag_h < k_box_select_min_axis_px)
414 {
415 box_select.active = false;
416 box_select.dragging = false;
417 return;
418 }
419
420 const bool additive = input.key_down(EditorKey::Shift);
421 const bool had_selection = !world_editor_state.selected_ids.empty();
422 auto ids_in_drag = gather_box_selection_ids(
423 world,
424 engine_context->simulation.physics,
425 camera_view_matrix,
426 camera_proj_matrix,
427 viewport_img_pos,
428 viewport_img_size,
429 drag_rect
430 );
431 const bool through_depth = imgui_io.KeyCtrl || imgui_io.KeySuper;
432 if (!through_depth)
433 {
434 ids_in_drag = filter_visible_box_selection_ids(
435 world,
436 ids_in_drag,
437 camera_view_matrix,
438 camera_proj_matrix,
439 viewport_img_pos,
440 viewport_img_size,
441 drag_rect
442 );
443 }
444
445 usize added_count{0zu};
446 if (!additive)
447 {
448 world_editor_state.selected_ids = ids_in_drag;
449 if (world_editor_state.selected_ids.empty())
450 {
451 world_editor_state.active_id.reset();
452 }
453 else
454 {
455 world_editor_state.active_id = world_editor_state.selected_ids.back();
456 }
457 added_count = world_editor_state.selected_ids.size();
458 }
459 else
460 {
461 for (const auto id : ids_in_drag)
462 {
463 if (!world_editor_state.is_selected(id))
464 {
465 world_editor_state.toggle_selection(id);
466 ++added_count;
467 }
468 }
469 }
470
471 if (added_count > 0zu)
472 {
473 if (additive)
474 {
475 ui_log(
476 std::format(
477 "Added {} cube{} to selection", added_count, plural_suffix(added_count)
478 )
479 );
480 }
481 else
482 {
483 ui_log(std::format("Selected {} cube{} (box)", added_count, plural_suffix(added_count)));
484 }
485 }
486 else if (!additive && had_selection)
487 {
488 ui_log("Deselected all");
489 }
490
491 box_select.active = false;
492 box_select.dragging = false;
493}

495auto GfxContext::hover_interaction_holding_middle(EditorInput& input, Camera& cam) const -> void
496{
497 const auto dx = input.ui_dx();
498 const auto dy = input.ui_dy();
499
500 if (input.key_down(EditorKey::Shift))
501 { // Move Pivot
502 const auto pan_world = cam.pan_offset_world(dx, dy, viewport_img_size.y);
503 cam.pivot += pan_world * k_pan_sensitivity;
504 }
505 else
506 { // Rotate Around pivot
507 constexpr bool k_invert_mouse_pivot_y{false};
508
509 cam.yaw += -dx * k_sensitivity;
510 cam.pitch += (k_invert_mouse_pivot_y ? -1.0f : 1.0f) * dy * k_sensitivity;
511
512 const auto pitch_lim = glm::radians(k_camera_pitch_lim_deg);
513 cam.pitch = std::clamp(cam.pitch, -pitch_lim, pitch_lim);
514 }
515}
