144auto GfxContext::begin_grab(const EditorInput& input) -> void
145{
146 Expects(engine_context);
147
148 const auto& selected_ids = engine_context->simulation.world.editor_state().selected_ids;
149 if (selected_ids.empty())
150 {
151 ui_log("Grab (G): nothing selected");
152 return;
153 }
154
155 auto& grab = editor.grab;
156
157 grab.active = true;
158 grab.start_mouse_x = input.ui_mouse_x;
159 grab.start_mouse_y = input.ui_mouse_y;
160 grab.constraint = EditorState::GrabConstraint::None;
161
162 grab.start_positions.clear();
163 grab.start_positions.reserve(selected_ids.size());
164
165 for (const EntityId id : selected_ids)
166 {
167 if (auto* entity = engine_context->simulation.world.find(id))
168 {
169 Pos3 start_pos = entity->transform.position;
170
171 if (entity->body)
172 {
173 if (auto rb = engine_context->simulation.physics.try_body(*entity->body))
174 {
175 rb->grabbed = true;
176 rb->asleep = false;
177 rb->sleep_frames = 0;
178 rb->velocity = Dir3{};
179 rb->angular_velocity = Dir3{};
180 start_pos = rb->position;
181 (void) engine_context->simulation.world.set_position(id, start_pos);
182 }
183 }
184
185 grab.start_positions.emplace_back(id, start_pos);
186 }
187 }
188
189 ui_log("Grab: move mouse. X/Y/Z constrain. LMB/Enter confirm. RMB/Esc cancel.");
190}

192auto GfxContext::update_grab(const EditorInput& input) -> void
193{
194 auto& grab = editor.grab;
195 if (!grab.active)
196 {
197 return;
198 }
199
200 Expects(engine_context);
201
202 if (const auto c = grab_constraint_from_key(input); c != EditorState::GrabConstraint::None)
203 {
204 set_grab_constraint(c);
205 }
206
207 const auto& cam = engine_context->simulation.world.editor_state().camera();
208
209 const auto vp_h = std::max(1.0f, viewport_img_size.y);
210 const auto units_per_px = (2.0f * cam.distance * std::tan(0.5f * cam.fov_y)) / vp_h;
211
212 const auto dx = narrow_cast<f32>(input.ui_mouse_x - grab.start_mouse_x);
213 const auto dy = narrow_cast<f32>(input.ui_mouse_y - grab.start_mouse_y);
214
215 auto delta = (dx * units_per_px) * cam.right() + (-dy * units_per_px) * cam.up();
216
217 using GC = EditorState::GrabConstraint;
218 switch (grab.constraint)
219 {
220 case GC::None:
221 break;
222 case GC::X:
223 delta = Dir3{delta.x, 0.0f, 0.0f};
224 break;
225 case GC::Y:
226 delta = Dir3{0.0f, delta.y, 0.0f};
227 break;
228 case GC::Z:
229 delta = Dir3{0.0f, 0.0f, delta.z};
230 break;
231 }
232
233 for (const auto& [id, start_pos] : grab.start_positions)
234 {
235 set_entity_and_body_position(engine_context->simulation, id, start_pos + delta);
236 }
237}
