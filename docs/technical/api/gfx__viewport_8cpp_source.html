31[[nodiscard]] auto compute_viewport_fb_rect(
32 GLFWwindow& window, const glm::vec2& img_pos_screen, const glm::vec2& img_size_screen
33) noexcept -> std::optional<RectInt>
34{
35 int fb_width{0}, fb_height{0};
36 glfwGetFramebufferSize(&window, &fb_width, &fb_height);
37
38 int win_width{0}, win_height{0};
39 glfwGetWindowSize(&window, &win_width, &win_height);
40
41 if (fb_width <= 0 || fb_height <= 0 || win_width <= 0 || win_height <= 0)
42 {
43 return std::nullopt;
44 }
45
46 const auto fb_width_f = static_cast<f32>(fb_width);
47 const auto fb_height_f = static_cast<f32>(fb_height);
48 const auto win_width_f = static_cast<f32>(win_width);
49 const auto win_height_f = static_cast<f32>(win_height);
50
51 const auto scale_x = fb_width_f / win_width_f;
52 const auto scale_y = fb_height_f / win_height_f;
53
54 const auto left_px = img_pos_screen.x * scale_x;
55 const auto width_px = img_size_screen.x * scale_x;
56
57 // ImGui screen coords: origin top-left; OpenGL framebuffer coords: origin bottom-left.
58 const auto bottom_px = fb_height_f - (img_pos_screen.y + img_size_screen.y) * scale_y;
59 const auto height_px = img_size_screen.y * scale_y;
60
61 const auto vx = static_cast<int>(std::lround(left_px));
62 const auto vw = static_cast<int>(std::lround(width_px));
63 const auto vy = static_cast<int>(std::lround(bottom_px));
64 const auto vh = static_cast<int>(std::lround(height_px));
65
66 RectInt r{};
67 r.x = std::clamp(vx, 0, fb_width - 1);
68 r.y = std::clamp(vy, 0, fb_height - 1);
69 r.w = std::clamp(vw, 1, fb_width - r.x);
70 r.h = std::clamp(vh, 1, fb_height - r.y);
71 return r;
72}

84[[nodiscard]] auto compute_entity_color(const GfxContext& gfx, const Entity& ent) noexcept -> Color3
85{
86 Expects(gfx.engine_context);
87 Color3 color{ent.color};
88
89 if (!gfx.phys_debug.enabled)
90 {
91 return color;
92 }
93
94 const EngineContext* e = gfx.engine_context;
95 if (!e)
96 {
97 return color;
98 }
99
100 const auto rb = try_entity_body(e->simulation, ent);
101 if (!rb || rb->is_static())
102 {
103 return color;
104 }
105
106 using CM = GfxContext::PhysicsDebugSettings::ColorMode;
107 switch (gfx.phys_debug.color_mode)
108 {
109 case CM::Diffuse:
110 return color;
111
112 case CM::SleepState:
113 {
114 auto f_sleep_min = static_cast<f32>(rb->sleep_frames) / 60.0f;
115 const auto t = rb->asleep ? 1.0f : std::clamp(f_sleep_min, 0.0f, 1.0f);
116 return mix(gfx.phys_debug.sleep_active_color, gfx.phys_debug.sleep_asleep_color, t);
117 }
118
119 case CM::KineticEnergy:
120 {
121 const auto m = 1.0f / rb->inv_mass;
122 auto E = 0.5f * m * glm::dot(rb->velocity, rb->velocity);
123 if (gfx.phys_debug.ke_include_angular)
124 {
125 // TODO: Should cache this
126 const glm::mat3 I_world = glm::inverse(rb->inv_inertia_world);
127 E += 0.5f * glm::dot(rb->angular_velocity, I_world * rb->angular_velocity);
128 }
129
130 const auto denom = std::max(1e-6f, gfx.phys_debug.ke_max);
131 const auto t = std::clamp(E / denom, 0.0f, 1.0f);
132
133 return mix(gfx.phys_debug.ke_low_color, gfx.phys_debug.ke_high_color, t);
134 }
135 }
136
137 return color;
138}

343auto GfxContext::render_to_viewport_outline(
344 const ViewMatrix& camera_view_matrix, const ProjMatrix& camera_proj_matrix
345) const -> void
346{
347 Expects(engine_context);
348 glEnable(GL_STENCIL_TEST);
349
350 const auto camera_pos = engine_context->simulation.world.editor_state().camera().position();
351
352 const auto& selected = engine_context->simulation.world.editor_state().selected_ids;
353 if (selected.empty())
354 {
355 return;
356 }
357
358 for (const EntityId id : selected)
359 {
360 const Entity* sel = engine_context->simulation.world.find(id);
361 if (!sel)
362 {
363 continue;
364 }
365
366 const auto sel_idx = engine_context->simulation.world.find_idx(id);
367 if (!sel_idx)
368 {
369 continue;
370 }
371 const auto model = engine_context->simulation.world.model_matrix_at(*sel_idx);
372 if (!model)
373 {
374 continue;
375 }
376
377 const auto M = ModelMatrix{*model};
378 glClear(GL_STENCIL_BUFFER_BIT);
379
380 { // Pass 1: write stencil (invisible)
381 glColorMask(GL_FALSE, GL_FALSE, GL_FALSE, GL_FALSE);
382 glDepthMask(GL_FALSE);
383 glDisable(GL_DEPTH_TEST);
384
385 glStencilMask(0xFF);
386 glStencilFunc(GL_ALWAYS, 1, 0xFF);
387 glStencilOp(GL_KEEP, GL_KEEP, GL_REPLACE);
388
389 const auto& prog = shader_programs.obj;
390 prog.bind();
391 prog.set_uView(camera_view_matrix);
392 prog.set_uProj(camera_proj_matrix);
393 prog.set_uModel(M);
394 prog.set_uCameraPos(camera_pos);
395 prog.set_uEnvLightStrength(k_env_light_strength);
396 prog.set_uEnvironmentTex(0);
397
398 glActiveTexture(GL_TEXTURE0);
399 glBindTexture(GL_TEXTURE_2D, textures.environment_lighting.id);
400
401 meshes.cube.instantiate_once();
402
403 glColorMask(GL_TRUE, GL_TRUE, GL_TRUE, GL_TRUE);
404 glDepthMask(GL_TRUE);
405 glEnable(GL_DEPTH_TEST);
406 glDepthFunc(GL_LESS);
407 }
408
409 { // Pass 2: draw outline where stencil != 1
410 glStencilFunc(GL_NOTEQUAL, 1, 0xFF);
411 glStencilMask(0x00);
412
413 glDisable(GL_DEPTH_TEST);
414 glDisable(GL_CULL_FACE);
415
416 const ModelMatrix M_outline{M.m * glm::scale(glm::mat4(1.0f), glm::vec3(1.04f))};
417
418 const auto& prog = shader_programs.outline;
419 prog.bind();
420 prog.set_uView(camera_view_matrix);
421 prog.set_uProj(camera_proj_matrix);
422 prog.set_uModel(M_outline);
423 prog.set_uColor(k_outline_color);
424
425 meshes.cube.instantiate_once();
426
427 glEnable(GL_DEPTH_TEST);
428 glDepthFunc(GL_LESS);
429
430 glStencilMask(0xFF);
431 glStencilFunc(GL_ALWAYS, 0, 0xFF);
432 glStencilOp(GL_KEEP, GL_KEEP, GL_REPLACE);
433 }
434 }

637auto GfxContext::append_selected_entity_debug_lines_() -> void
638{
639 Expects(engine_context);
640 const auto& world = engine_context->simulation.world;
641 const auto& physics = engine_context->simulation.physics;
642
643 const auto& selected = world.editor_state().selected_ids;
644 for (const EntityId id : selected)
645 {
646 const Entity* entity = world.find(id);
647 if (!entity)
648 {
649 continue;
650 }
651
652 const auto rigid_body = entity->body ? physics.try_body(*entity->body)
653 : std::optional<PhysicsContext::BodyConstRef>{};
654
655 const auto [center_of_mass, ori, extent] = [&]() -> std::tuple<Pos3, Quaternion, f32>
656 {
657 if (rigid_body)
658 {
659 return {
660 rigid_body->position,
661 rigid_body->orientation,
662 2.0f * max_extent(rigid_body->half_extents),
663 };
664 }
665
666 return {
667 entity->transform.position,
668 entity->transform.orientation,
669 max_extent(entity->transform.scale),
670 };
671 }();
672
673 const auto alpha = 1.0f;
674 if (phys_debug.show_selected_axes)
675 {
676 const auto axis_len = std::max(0.0f, extent) * std::max(0.0f, phys_debug.axis_scale);
677
678 const auto R = glm::mat3_cast(ori);
679
680 const auto ax = axis_len * glm::normalize(R * k_axis_x);
681 const auto ay = axis_len * glm::normalize(R * k_axis_y);
682 const auto az = axis_len * glm::normalize(R * k_axis_z);
683
684 push_debug_line_({center_of_mass, center_of_mass + ax}, Color3::Red, alpha);
685 push_debug_line_({center_of_mass, center_of_mass + ay}, Color3::Green, alpha);
