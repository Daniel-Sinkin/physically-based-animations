23struct RollingFrameStats
24{
25 static constexpr usize k_window_samples{512zu};
26
27 struct Snapshot
28 {
29 usize sample_count{};
30 f64 window_seconds{};
31 f64 avg_frame_ms{};
32 f64 fps{};
33 f64 p50_frame_ms{};
34 f64 p95_frame_ms{};
35 f64 p99_frame_ms{};
36 f64 max_frame_ms{};
37 bool valid{false};
38 };
39
40 std::array<f32, k_window_samples> samples_ms{};
41 usize next_write{0zu};
42 usize sample_count{0zu};
43 TimePoint last_frame_time{};
44 bool has_last_frame_time{false};
45
46 auto tick(TimePoint now) noexcept -> void
47 {
48 if (!has_last_frame_time)
49 {
50 last_frame_time = now;
51 has_last_frame_time = true;
52 return;
53 }
54
55 const auto dt_ms = std::chrono::duration<f64, std::milli>(now - last_frame_time).count();
56 last_frame_time = now;
57
58 if (!std::isfinite(dt_ms) || dt_ms <= 0.0)
59 {
60 return;
61 }
62
63 samples_ms[next_write] = static_cast<f32>(dt_ms);
64 next_write = (next_write + 1zu) % k_window_samples;
65 sample_count = std::min(sample_count + 1zu, k_window_samples);
66 }
67
68 [[nodiscard]] auto snapshot() const -> Snapshot
69 {
70 Snapshot out{};
71 out.sample_count = sample_count;
72 if (sample_count == 0zu)
73 {
74 return out;
75 }
76
77 auto ordered = std::vector<f32>{};
78 ordered.reserve(sample_count);
79
80 const auto oldest = (next_write + k_window_samples - sample_count) % k_window_samples;
81 for (auto i = 0zu; i < sample_count; ++i)
82 {
83 const auto idx = (oldest + i) % k_window_samples;
84 ordered.push_back(samples_ms[idx]);
85 }
86
87 auto sum_ms = 0.0;
88 for (const auto frame_ms : ordered)
89 {
90 sum_ms += static_cast<f64>(frame_ms);
91 }
92
93 out.window_seconds = sum_ms / 1000.0;
94 out.avg_frame_ms = sum_ms / static_cast<f64>(sample_count);
95 out.fps = (out.avg_frame_ms > 0.0) ? (1000.0 / out.avg_frame_ms) : 0.0;
96
97 std::sort(ordered.begin(), ordered.end());
98
99 auto percentile = [&](f64 p) -> f64
100 {
101 Expects(!ordered.empty());
102 const auto n = static_cast<f64>(ordered.size());
103 const auto rank = std::ceil(p * n);
104 const auto rank_1 = std::max(1.0, rank);
105 const auto idx = static_cast<usize>(rank_1 - 1.0);
106 return static_cast<f64>(ordered[std::min(idx, ordered.size() - 1zu)]);
107 };
108
109 out.p50_frame_ms = percentile(0.50);
110 out.p95_frame_ms = percentile(0.95);
111 out.p99_frame_ms = percentile(0.99);
112 out.max_frame_ms = static_cast<f64>(ordered.back());
113 out.valid = true;
114 return out;
115 }
116};

368constexpr auto rgba_u32(u32 rgba) noexcept -> ImVec4
369{
370 constexpr float inv{1.0f / 255.0f};
371 const auto r = static_cast<float>((rgba >> 24) & 0xFFu);
372 const auto g = static_cast<float>((rgba >> 16) & 0xFFu);
373 const auto b = static_cast<float>((rgba >> 8) & 0xFFu);
374 const auto a = static_cast<float>((rgba >> 0) & 0xFFu);
375 return ImVec4(r * inv, g * inv, b * inv, a * inv);
376}
